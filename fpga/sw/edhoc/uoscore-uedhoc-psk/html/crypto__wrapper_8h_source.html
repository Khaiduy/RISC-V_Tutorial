147#ifdef EDHOC_MOCK_CRYPTO_WRAPPER
148/*
149 * Elliptic curve based signature algorithms generate signatures that are not
150 * deterministic. In order to test edhoc module against test vectors provided
151 * by the RFC authors, a mocking functionality has been added.
152 *
153 * When EDHOC_MOCK_CRYPTO_WRAPPER macro is defined, structure
154 * edhoc_crypto_mock_cb can be used to define values returned/generated by
155 * the sign() and aead() functions. Predefined value will be used only if the
156 * function has been called with arguments values matching those provided in
157 * edhoc_crypto_mock_cb.aead_in_out / edhoc_crypto_mock_cb.sign_in_out structure.
158 *
159 * When there is no matching arguments, the function aead()/sign() will
160 * continue normally.
161 */
162struct edhoc_mock_aead_in_out {
163 struct byte_array out;
164 struct byte_array in;
165 struct byte_array key;
166 struct byte_array nonce;
167 struct byte_array aad;
168 struct byte_array tag;
169};
170
171struct edhoc_mock_sign_in_out {
172 enum sign_alg curve;
173 struct byte_array sk;
174 struct byte_array pk;
175 struct byte_array msg;
176 struct byte_array out;
177};

180 int aead_in_out_count;

183 struct edhoc_mock_sign_in_out *sign_in_out;
184};
